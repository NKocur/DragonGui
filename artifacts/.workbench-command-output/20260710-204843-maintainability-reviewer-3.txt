  88: def frame_from_columns(**columns: Any) -> dict[str, list[Any]]:

  92: def build_scatter_frame(count: int = POINTS, phase: float = 0.0) -> dict[str, list[Any]]:
  93:     t = np.linspace(0, 9 * math.pi, count, dtype=np.float32)
  94:     ring = 1.0 + 0.24 * np.sin(t * 0.37 + phase)
  95:     x = ring * np.cos(t) + RNG.normal(0, 0.055, count)
  96:     y = ring * np.sin(t) + RNG.normal(0, 0.055, count)
  97:     z = np.linspace(-1.4, 1.4, count, dtype=np.float32) + 0.32 * np.sin(t * 0.6 + phase)
  98:     health = np.clip(82 + 11 * np.sin(t * 0.19 + phase) + RNG.normal(0, 3, count), 0, 100)
  99:     temp = np.clip(58 + 16 * np.cos(t * 0.11 - phase) + RNG.normal(0, 2.5, count), 15, 95)
 100:     return frame_from_columns(
 101:         x=x,
 102:         y=y,
 103:         z=z,
 104:         health=health,
 105:         temperature=temp,
 106:         zone=[f"zone-{i % 8}" for i in range(count)],
 107:         asset=[f"cell-{(i * 7) % 19:02d}" for i in range(count)],
 108:     )

 111: def build_table(rows: int = ROWS) -> list[dict[str, Any]]:
 112:     owners = ["ingest", "model", "quality", "export", "field"]
 113:     states = ["ready", "running", "queued", "watch", "blocked"]
 114:     table: list[dict[str, Any]] = []
 115:     now = datetime(2026, 7, 10, 9, 0)
 116:     for i in range(rows):
 117:         latency = 18 + (i % 17) * 2.6 + (i % 5) * 0.7
 118:         error_rate = round(max(0, math.sin(i * 0.21) * 1.8 + (i % 9) * 0.17), 2)
 119:         state_name = states[(i + (1 if error_rate > 1.8 else 0)) % len(states)]
 120:         table.append(
 121:             {
 122:                 "route": f"plant.north.line-{i % 6}.cell-{i % 14}.sensor-feed",
 123:                 "owner": owners[i % len(owners)],
 124:                 "state": state_name,
 125:                 "latency_ms": round(latency, 2),
 126:                 "error_rate": error_rate,
 127:                 "throughput": 950 + (i * 37) % 760,
 128:                 "updated": (now - timedelta(minutes=i * 3)).isoformat(timespec="minutes"),
 129:             }
 130:         )
 131:     return table

 146:     y_labels = [f"L{i + 1}" for i in range(8)]

 157: def write_png(path: Path, width: int = 420, height: int = 220) -> None:
 158:     rows = []
 159:     for y in range(height):
 160:         raw = bytearray([0])
 161:         for x in range(width):
 162:             signal = int(80 + 70 * math.sin((x + y) / 33.0))
 163:             raw.extend((24 + x * 70 // width, 70 + signal // 3, 92 + y * 95 // height))
 164:         rows.append(bytes(raw))
 165:     payload = zlib.compress(b"".join(rows), 9)
 166: 
 167:     def chunk(kind: bytes, data: bytes) -> bytes:
 168:         return struct.pack(">I", len(data)) + kind + data + struct.pack(">I", zlib.crc32(kind + data) & 0xFFFFFFFF)
 169: 
 170:     png = (
 171:         b"\x89PNG\r\n\x1a\n"
 172:         + chunk(b"IHDR", struct.pack(">IIBBBBB", width, height, 8, 2, 0, 0, 0))
 173:         + chunk(b"IDAT", payload)
 174:         + chunk(b"IEND", b"")
 175:     )
 176:     path.write_bytes(png)
