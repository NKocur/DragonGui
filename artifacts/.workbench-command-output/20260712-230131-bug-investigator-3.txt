   84: fn display_list_color(value: Option<&Value>, theme: &Theme, fallback: Color) -> Color {
   85:     match value {
   86:         Some(Value::String(text)) => parse_web_color(text)
   87:             .or_else(|| parse_hex_color(text))
   88:             .unwrap_or_else(|| ColorRef::Token(text.trim().to_string()).resolve(theme)),
   89:         Some(Value::Array(items)) if items.len() == 3 || items.len() == 4 => {
   90:             let r = items.first().and_then(value_f32).unwrap_or(0.0);
   91:             let g = items.get(1).and_then(value_f32).unwrap_or(0.0);
   92:             let b = items.get(2).and_then(value_f32).unwrap_or(0.0);
   93:             let a = items.get(3).and_then(value_f32).unwrap_or(1.0);
   94:             [
   95:                 normalize_color_channel(r),
   96:                 normalize_color_channel(g),
   97:                 normalize_color_channel(b),
   98:                 normalize_color_channel(a),
   99:             ]
  100:         }
  101:         _ => fallback,
  102:     }
  103: }

  105: fn display_list_scale(node: &WidgetNode, rect: [f32; 4]) -> (f32, f32) {
  106:     let paint_w = node

  114:     let paint_h = node

  120:         .or(node.props.intrinsic_height)
  121:         .unwrap_or(rect[3].max(1.0));

  125: fn emit_extension_display_list(

  132:         return;

  135:     let stroke_scale = ((sx.abs() + sy.abs()) * 0.5).max(0.001);
  136:     for command in commands {
  137:         let Some(command) = command.as_object() else {

  151:                     continue;
  152:                 };

  154:                     object_f32(command, "w").or_else(|| object_f32(command, "width"))
  155:                 else {
  156:                     continue;
  157:                 };
  158:                 let Some(local_h) =
  159:                     object_f32(command, "h").or_else(|| object_f32(command, "height"))

  161:                     continue;
  162:                 };
  163:                 if local_w <= 0.0 || local_h <= 0.0 {
  164:                     continue;
  165:                 }

  167:                     object_f32(command, "radius").unwrap_or(0.0).max(0.0) * sx.min(sy).abs();
  168:                 let screen = [
  169:                     rect[0] + local_x * sx,
  170:                     rect[1] + local_y * sy,
  171:                     local_w * sx,
  172:                     local_h * sy,
  173:                 ];

  177:                 }

  181:                         .or_else(|| object_f32(command, "line_width"))

  193:                     }
  194:                 }

  199:                 };

  201:                     continue;

  204:                     continue;

  207:                     continue;
  208:                 };

  210:                 let end = [rect[0] + x2 * sx, rect[1] + y2 * sy];
  211:                 let stroke = display_list_color(command.get("stroke"), theme, theme.accent);
  212:                 let width = object_f32(command, "stroke_width")
  213:                     .or_else(|| object_f32(command, "width"))
  214:                     .unwrap_or(1.5)
  215:                     .max(0.1)

  218:                     push_line_segment(out, a, b, width, stroke);

  223:                     continue;
  224:                 };
  225:                 if points.len() < 2 {
  226:                     continue;
  227:                 }
  228:                 let stroke = display_list_color(command.get("stroke"), theme, theme.accent);
  229:                 let width = object_f32(command, "stroke_width")
  230:                     .or_else(|| object_f32(command, "width"))
  231:                     .unwrap_or(1.5)
  232:                     .max(0.1)

  257:                     }
  258:                     previous = Some(current);
  259:                 }

 1271:     fn render_batches(&self, pass: &mut wgpu::RenderPass<'_>, batches: &[PrimitiveBatch]) {
 1272:         let mut current_kind = None;
 1273:         for batch in batches {
 1274:             if batch.count == 0 {
 1275:                 continue;
 1276:             }
 1277:             if current_kind != Some(batch.kind) {
 1278:                 match batch.kind {
 1279:                     PrimitivePipelineKind::Simple => {
 1280:                         pass.set_pipeline(&self.simple_pipeline);
 1281:                         pass.set_bind_group(0, &self.bind_group, &[]);
 1282:                         pass.set_vertex_buffer(0, self.simple_vertex_buffer.slice(..));
 1283:                     }
 1284:                     PrimitivePipelineKind::Line => {
 1285:                         pass.set_pipeline(&self.line_pipeline);
 1286:                         pass.set_bind_group(0, &self.bind_group, &[]);
 1287:                         pass.set_vertex_buffer(0, self.line_vertex_buffer.slice(..));
 1288:                     }
 1289:                     PrimitivePipelineKind::Complex => {
 1290:                         pass.set_pipeline(&self.complex_pipeline);
 1291:                         pass.set_bind_group(0, &self.bind_group, &[]);
 1292:                         pass.set_vertex_buffer(0, self.complex_vertex_buffer.slice(..));
 1293:                     }
 1294:                 }
 1295:                 current_kind = Some(batch.kind);
 1296:             }
 1297:             pass.draw(0..6, batch.start..batch.start + batch.count);
 1298:         }
 1299:     }

 3226:             }
