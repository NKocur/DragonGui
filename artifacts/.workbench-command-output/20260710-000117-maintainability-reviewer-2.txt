404:def inline_badge_bounds_violation(
405:    node_id: str,
406:    badge: str,
407:    node_type: str,
408:    rect: dict[str, Any],
409:    computed_style: Any,
410:    spacing: float,
411:    theme_font_size: float,
412:) -> str | None:
413:    try:
414:        x = float(rect["x"])
415:        y = float(rect["y"])
416:        w = float(rect["w"])
417:        h = float(rect["h"])
418:    except (KeyError, TypeError, ValueError):
419:        return f"{node_id}: malformed rect"
420:    style = computed_style.get("style") if isinstance(computed_style, dict) else {}
421:    if not isinstance(style, dict):
422:        style = {}
423:    text_style = style.get("text") if isinstance(style.get("text"), dict) else {}
424:    parts = style.get("parts") if isinstance(style.get("parts"), dict) else {}
425:    badge_part = parts.get("badge") if isinstance(parts.get("badge"), dict) else {}
426:    badge_text = badge_part.get("text") if isinstance(badge_part.get("text"), dict) else {}
427:    badge_visual = badge_part.get("visual") if isinstance(badge_part.get("visual"), dict) else {}
428:
429:    badge_font = float(
430:        badge_text.get("font_size")
431:        or text_style.get("font_size")
432:        or max(theme_font_size - 2.0, 10.0)
433:    )
434:    border = max(0.0, float(badge_visual.get("border_width") or 0.0))
435:    preferred_w = max(len(badge) * badge_font * 0.68 + 8.0 * 2.0 + border * 2.0, 16.0)
436:    preferred_h = max(badge_font + 6.0 + border * 2.0, 16.0)
437:    right_inset = 8.0 if node_type == "tab" else spacing
438:    badge_w = min(preferred_w, max(w - right_inset, 0.0))
439:    badge_h = min(preferred_h, max(h - 4.0, 1.0))
440:    badge_x = x + w - right_inset - badge_w
441:    badge_y = y + (h - badge_h) * 0.5
442:
443:    if badge_w <= 0.0 or badge_h <= 0.0:
444:        return f"{node_id}: badge has no visible area"
445:    if badge_x < x - 0.5 or badge_y < y - 0.5:
446:        return f"{node_id}: badge starts outside parent"
447:    if badge_x + badge_w > x + w + 0.5 or badge_y + badge_h > y + h + 0.5:
448:        return f"{node_id}: badge exceeds parent"
449:    return None
