class DenseScatter3DFrame:
    columns = ("x", "y", "z", "score", "radius")
    dtypes = ("float32", "float32", "float32", "float32", "float32")

    def __init__(self, rows: int = POINTS) -> None:
        self.shape = (rows, len(self.columns))
        i = np.arange(rows, dtype=np.float32)
        t = (i + np.float32(0.5)) / np.float32(max(rows, 1))
        golden_angle = np.float32(math.pi * (3.0 - math.sqrt(5.0)))
        theta = i * golden_angle

        z_unit = np.float32(1.0) - np.float32(2.0) * t
        shell_radius = np.sqrt(np.maximum(np.float32(0.0), np.float32(1.0) - z_unit * z_unit))
        fill = np.sqrt(np.mod(i * np.float32(0.7548777), np.float32(1.0)))
        radius = (shell_radius * fill) * np.float32(3.1)

        self.x = np.cos(theta) * radius
        self.y = np.sin(theta) * radius
        self.z = z_unit * np.float32(2.65) + np.sin(theta * np.float32(0.37)) * np.float32(0.18)
        self.radius = np.sqrt(self.x * self.x + self.y * self.y + self.z * self.z).astype(np.float32)
        wave = (np.sin(theta * np.float32(0.16) + self.z) + np.float32(1.0)) * np.float32(0.5)
        self.score = (wave * np.float32(0.42) + t * np.float32(0.58)).astype(np.float32)

    def __getitem__(self, column: str) -> object:
        return getattr(self, column)

frame = DenseScatter3DFrame()
stop_event = threading.Event()
runner_thread: threading.Thread | None = None
state_lock = threading.Lock()
state = {
    "running": False,
    "submitted": 0,
    "elapsed_ms": 0.0,
}
metrics: dg.Label | None = None


def camera_state(index: int) -> dict[str, object]:
    phase = index / max(CAMERA_UPDATES, 1)
    return {
        "target": [0.0, 0.0, 0.0],
        "distance": 8.5,
        "yaw": 0.35 + phase * math.tau * 1.5,
        "pitch": 0.42 + math.sin(phase * math.tau * 2.0) * 0.35,
        "parallel": False,
    }


def set_metrics(text: str) -> None:
    if metrics is not None:
        metrics.set_value(text)


def update_metrics() -> None:
    with state_lock:
        values = dict(state)
    rate = values["submitted"] / (values["elapsed_ms"] / 1000.0) if values["elapsed_ms"] else 0.0
    set_metrics(
        "\n".join(
            [
                f"points: {POINTS:,}",
                f"camera updates: {values['submitted']}/{CAMERA_UPDATES}",
                f"interval: {CAMERA_INTERVAL_MS:.1f} ms",
                f"elapsed: {values['elapsed_ms']:.2f} ms",
                f"submitted: {rate:.1f}/s",
            ]
        )
    )


def run_camera_updates() -> None:
    started = time.perf_counter()
    with state_lock:
        state["running"] = True
        state["submitted"] = 0
        state["elapsed_ms"] = 0.0
    update_metrics()

    for index in range(CAMERA_UPDATES):
        if stop_event.is_set():
            break
        try:
            app.call_soon_threadsafe(lambda i=index: dense_plot.set_camera(camera_state(i)))
        except RuntimeError:
            break
        with state_lock:
            state["submitted"] = index + 1
            state["elapsed_ms"] = (time.perf_counter() - started) * 1000.0
        if CAMERA_INTERVAL_MS > 0.0 and stop_event.wait(CAMERA_INTERVAL_MS / 1000.0):
            break

    if EXIT_ON_DONE:
        try:
            app.debug_snapshot(timeout_ms=5000)
        except RuntimeError:
            pass
        try:
            app.request_exit()
        except RuntimeError:
            pass

    with state_lock:
        state["running"] = False
        state["elapsed_ms"] = (time.perf_counter() - started) * 1000.0
    update_metrics()


def start_camera_benchmark() -> None:
    global runner_thread
    if runner_thread is not None and runner_thread.is_alive():
        return
    stop_event.clear()
    runner_thread = threading.Thread(target=run_camera_updates, daemon=True)
    runner_thread.start()


def stop_camera_benchmark() -> None:
    stop_event.set()
    with state_lock:
        state["running"] = False
    update_metrics()
