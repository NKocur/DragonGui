 4108:     fn scaled_local_scissor(&self, target_width: u32, target_height: u32) -> ([u32; 2], [u32; 2]) {
 4109:         if self.width == 0 || self.height == 0 || target_width == 0 || target_height == 0 {
 4110:             return ([0, 0], [0, 0]);
 4111:         }
 4112:         let sx = target_width as f32 / self.width as f32;
 4113:         let sy = target_height as f32 / self.height as f32;
 4114:         let left = ((self.scissor_offset[0] as f32 - self.offset[0]).max(0.0) * sx)
