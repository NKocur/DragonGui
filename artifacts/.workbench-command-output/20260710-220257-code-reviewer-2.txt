 3425: fn scroll_content_bounds(node: &WidgetNode, result: &LayoutResult) -> Option<ScrollContentBounds> {
 3426:     let mut right = f32::NEG_INFINITY;
 3427:     let mut top = f32::INFINITY;
 3428:     let mut bottom = f32::NEG_INFINITY;
 3429:     for child in &node.children {
 3430:         scroll_content_bounds_for_child(child, result, &mut right, &mut top, &mut bottom);
 3431:     }
 3432:     if top.is_finite() {
 3433:         let scale_factor = if result.scale_factor > 0.0 {
 3434:             result.scale_factor
 3435:         } else {
 3436:             1.0
 3437:         };
 3438:         right += scroll_container_right_padding_lp(node) * scale_factor;
 3439:         bottom += scroll_container_bottom_padding_lp(node) * scale_factor;
 3440:         Some(ScrollContentBounds { right, bottom })
 3441:     } else {
 3442:         None
 3443:     }
 3444: }

 3446: fn scroll_content_bounds_for_child(
 3447:     node: &WidgetNode,
 3448:     result: &LayoutResult,
 3449:     right: &mut f32,
 3450:     top: &mut f32,
 3451:     bottom: &mut f32,
 3452: ) {
 3453:     if is_fixed_positioned_node(node) {
 3454:         return;
 3455:     }
 3456:     let Some(rect) = result.rects.get(&node.id).copied() else {
 3457:         return;
 3458:     };
 3459:     if rect.w <= 0.0 || rect.h <= 0.0 {
 3460:         return;
 3461:     }
 3462: 
 3463:     *right = right.max(rect.x + rect.w);
 3464:     *top = top.min(rect.y);
 3465:     *bottom = bottom.max(rect.y + rect.h);
 3466: 
 3467:     if subtree_scroll_bounds_stop_at_node(node) {
 3468:         return;
 3469:     }
 3470:     for child in &node.children {
 3471:         scroll_content_bounds_for_child(child, result, right, top, bottom);
 3472:     }
 3473: }

 3480:         node_overflow_x(node),

 3482:     ) || matches!(
 3483:         node_overflow_y(node),
 3484:         Some(OverflowStyle::Hidden | OverflowStyle::Scroll | OverflowStyle::Auto)
 3485:     )
 3486: }
