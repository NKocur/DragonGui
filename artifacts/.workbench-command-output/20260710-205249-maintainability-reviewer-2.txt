  86: def frame_from_columns(**columns: Any) -> dict[str, list[Any]]:
  87:     return {name: list(values) for name, values in columns.items()}

  90: def build_scatter_frame(
  91:     count: int = POINTS,
  92:     phase: float = 0.0,
  93:     *,
  94:     rng: np.random.Generator | None = None,
  95: ) -> dict[str, list[Any]]:
  96:     rng = rng or np.random.default_rng(42)
  97:     t = np.linspace(0, 9 * math.pi, count, dtype=np.float32)
  98:     ring = 1.0 + 0.24 * np.sin(t * 0.37 + phase)
  99:     x = ring * np.cos(t) + rng.normal(0, 0.055, count)
 100:     y = ring * np.sin(t) + rng.normal(0, 0.055, count)
 101:     z = np.linspace(-1.4, 1.4, count, dtype=np.float32) + 0.32 * np.sin(t * 0.6 + phase)
 102:     health = np.clip(82 + 11 * np.sin(t * 0.19 + phase) + rng.normal(0, 3, count), 0, 100)
 103:     temp = np.clip(58 + 16 * np.cos(t * 0.11 - phase) + rng.normal(0, 2.5, count), 15, 95)
 104:     return frame_from_columns(
 105:         x=x,
 106:         y=y,
 107:         z=z,
 108:         health=health,
 109:         temperature=temp,
 110:         zone=[f"zone-{i % 8}" for i in range(count)],
 111:         asset=[f"cell-{(i * 7) % 19:02d}" for i in range(count)],
 112:     )

 115: def build_table(rows: int = ROWS) -> list[dict[str, Any]]:
 116:     owners = ["ingest", "model", "quality", "export", "field"]
 117:     states = ["ready", "running", "queued", "watch", "blocked"]
 118:     table: list[dict[str, Any]] = []
 119:     now = datetime(2026, 7, 10, 9, 0)
 120:     for i in range(rows):
 121:         latency = 18 + (i % 17) * 2.6 + (i % 5) * 0.7
 122:         error_rate = round(max(0, math.sin(i * 0.21) * 1.8 + (i % 9) * 0.17), 2)
 123:         state_name = states[(i + (1 if error_rate > 1.8 else 0)) % len(states)]
 124:         table.append(
 125:             {
 126:                 "route": f"plant.north.line-{i % 6}.cell-{i % 14}.sensor-feed",
 127:                 "owner": owners[i % len(owners)],
 128:                 "state": state_name,
 129:                 "latency_ms": round(latency, 2),
 130:                 "error_rate": error_rate,
 131:                 "throughput": 950 + (i * 37) % 760,
 132:                 "updated": (now - timedelta(minutes=i * 3)).isoformat(timespec="minutes"),
 133:             }
 134:         )
 135:     return table

 138: def build_series(points: int = SERIES_POINTS, *, rng: np.random.Generator | None = None) -> dict[str, list[float]]:
 139:     rng = rng or np.random.default_rng(42)

 146:         error_budget=np.clip(92 - np.abs(base) * 2 + rng.normal(0, 1.2, points), 65, 100),

 150: def build_heatmap() -> tuple[list[list[float]], list[str], list[str]]:
 151:     y_labels = [f"L{i + 1}" for i in range(8)]
 152:     x_labels = [f"{hour:02d}:00" for hour in range(0, 24, 2)]
 153:     matrix = []
 154:     for y in range(len(y_labels)):
 155:         row = []
 156:         for x in range(len(x_labels)):
 157:             row.append(18 + 8 * math.sin(x / 2.4) + 3.5 * math.cos(y * 0.8) + (x * y) % 4)

 159:     return matrix, x_labels, y_labels

 162: def write_png(path: Path, width: int = 420, height: int = 220) -> None:
 163:     rows = []
 164:     for y in range(height):
 165:         raw = bytearray([0])
 166:         for x in range(width):
 167:             signal = int(80 + 70 * math.sin((x + y) / 33.0))
 168:             raw.extend((24 + x * 70 // width, 70 + signal // 3, 92 + y * 95 // height))
 169:         rows.append(bytes(raw))
 170:     payload = zlib.compress(b"".join(rows), 9)
 171: 
 172:     def chunk(kind: bytes, data: bytes) -> bytes:
 173:         return struct.pack(">I", len(data)) + kind + data + struct.pack(">I", zlib.crc32(kind + data) & 0xFFFFFFFF)
 174: 
 175:     png = (
 176:         b"\x89PNG\r\n\x1a\n"
 177:         + chunk(b"IHDR", struct.pack(">IIBBBBB", width, height, 8, 2, 0, 0, 0))
 178:         + chunk(b"IDAT", payload)
 179:         + chunk(b"IEND", b"")
 180:     )
 181:     path.write_bytes(png)
