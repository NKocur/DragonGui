                other_label=other_label,
            )
        else:
            if labels is None or values is None:

            payload = _normalize_pie_data(

                values,
                colors=colors,
                top_n=top_n,
                other_label=other_label,
            )

        self.aggregate = _pie_aggregate(aggregate)

        self.other_label = str(other_label)
