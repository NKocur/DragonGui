class POINT(ctypes.Structure):
    _fields_ = [("x", ctypes.c_long), ("y", ctypes.c_long)]


class LeftFrame:
    columns = ("x", "y", "z", "temperature", "label")
    dtypes = ("float32", "float32", "float32", "float32", "str")

    def __init__(self, rows: int = 720) -> None:
        t = np.linspace(0.0, 1.0, rows, dtype=np.float32)
        theta = t * np.float32(np.pi * 7.5)
        radius = np.float32(0.35) + t * np.float32(2.8)
        self.x = np.cos(theta) * radius - np.float32(8.0)
        self.y = np.sin(theta) * radius + np.float32(1.0)
        self.z = (t - np.float32(0.5)) * np.float32(5.0)
        self.temperature = (np.float32(45.0) + t * np.float32(18.0)).astype(np.float32)
        self.label = np.array([f"left-{i}" for i in range(rows)], dtype=object)
        self.shape = (rows, len(self.columns))

    def __getitem__(self, column: str) -> object:
        return getattr(self, column)


class RightFrame:
    columns = ("x", "y", "z", "zone", "label")
    dtypes = ("float32", "float32", "float32", "float32", "str")

    def __init__(self, rows: int = 360) -> None:
        t = np.linspace(0.0, 1.0, rows, dtype=np.float32)
        self.x = np.float32(120.0) + t * np.float32(25.0)
        self.y = np.float32(210.0) + np.sin(t * np.float32(np.pi * 5.0)) * np.float32(18.0)
        self.z = np.zeros(rows, dtype=np.float32)
        self.zone = np.where(t > 0.5, np.float32(1.0), np.float32(0.0))
        self.label = np.array([f"right-{i}" for i in range(rows)], dtype=object)
        self.shape = (rows, len(self.columns))

    def __getitem__(self, column: str) -> object:
        return getattr(self, column)


def find_probe_window() -> int | None:
    user32 = ctypes.windll.user32
    pid = os.getpid()
    found = ctypes.c_void_p(0)

    @ctypes.WINFUNCTYPE(ctypes.c_bool, ctypes.c_void_p, ctypes.c_void_p)
    def enum_proc(hwnd: int, _param: int) -> bool:
        if not user32.IsWindowVisible(hwnd):
            return True
        window_pid = ctypes.c_ulong()
        user32.GetWindowThreadProcessId(hwnd, ctypes.byref(window_pid))
        if window_pid.value != pid:
            return True
        length = user32.GetWindowTextLengthW(hwnd)
        if length <= 0:
            return True
        buf = ctypes.create_unicode_buffer(length + 1)
        user32.GetWindowTextW(hwnd, buf, length + 1)
        if TITLE in buf.value:
            found.value = hwnd
            return False
        return True

    user32.EnumWindows(enum_proc, 0)
    return int(found.value) if found.value else None


def client_to_screen(hwnd: int, x: float, y: float) -> tuple[int, int]:
    pt = POINT(int(round(x)), int(round(y)))
    ctypes.windll.user32.ClientToScreen(hwnd, ctypes.byref(pt))
    return int(pt.x), int(pt.y)


def send_mouse_move(hwnd: int, x: float, y: float) -> None:
    sx, sy = client_to_screen(hwnd, x, y)
    ctypes.windll.user32.SetCursorPos(sx, sy)
    time.sleep(0.04)


def mouse_event(flags: int, data: int = 0) -> None:
    ctypes.windll.user32.mouse_event(flags, 0, 0, data, 0)
    time.sleep(0.05)


def click(hwnd: int, x: float, y: float) -> None:
    send_mouse_move(hwnd, x, y)
    mouse_event(0x0002)
    mouse_event(0x0004)


def drag(hwnd: int, start: tuple[float, float], end: tuple[float, float]) -> None:
    send_mouse_move(hwnd, *start)
    mouse_event(0x0002)
    send_mouse_move(hwnd, *end)
    mouse_event(0x0004)


def drag_outside_window(hwnd: int, start: tuple[float, float]) -> None:
    send_mouse_move(hwnd, *start)
    mouse_event(0x0002)
    sx, sy = client_to_screen(hwnd, -80, -80)
    ctypes.windll.user32.SetCursorPos(sx, sy)
    time.sleep(0.12)
    mouse_event(0x0004)
    time.sleep(0.08)


def wheel(hwnd: int, x: float, y: float, delta: int) -> None:
    send_mouse_move(hwnd, x, y)
    mouse_event(0x0800, delta)


def scatter_metrics(snapshot: dict[str, object], scatter_id: str) -> dict[str, object]:
    scatters = (
        snapshot.get("gpu", {})
        .get("resources", {})
        .get("scatters", {})
    )
    if not isinstance(scatters, dict) or scatter_id not in scatters:
        raise RuntimeError(f"scatter {scatter_id!r} missing from debug snapshot")
    value = scatters[scatter_id]
    if not isinstance(value, dict):
        raise RuntimeError(f"scatter {scatter_id!r} snapshot is not an object")
    return value


def viewport(snapshot: dict[str, object], scatter_id: str) -> dict[str, object]:
    metrics = scatter_metrics(snapshot, scatter_id)
    value = metrics.get("viewport")
    if not isinstance(value, dict):
        raise RuntimeError(f"scatter {scatter_id!r} snapshot has no viewport detail")
    return value


def viewport_center(snapshot: dict[str, object], scatter_id: str) -> tuple[float, float]:
    view = viewport(snapshot, scatter_id)
    offset = view.get("offset")
    size = view.get("size")
    if not isinstance(offset, list) or not isinstance(size, list):
        raise RuntimeError(f"scatter {scatter_id!r} viewport detail is malformed")
    return float(offset[0]) + float(size[0]) * 0.5, float(offset[1]) + float(size[1]) * 0.5


def stable_signature(snapshot: dict[str, object], scatter_id: str) -> dict[str, object]:
    metrics = scatter_metrics(snapshot, scatter_id)
    return {
        "last_point_count": metrics.get("last_point_count"),
        "primary_compact_colormap": metrics.get("primary_compact_colormap"),
        "scalar_bar": metrics.get("scalar_bar"),
        "bounds": metrics.get("bounds"),
        "actors": metrics.get("actors"),
        "payload_format": metrics.get("payload_format"),
    }


def command_tail(snapshot: dict[str, object]) -> list[dict[str, object]]:
    recent = snapshot.get("runtime", {}).get("commands", {}).get("recent", [])
    return recent[-18:] if isinstance(recent, list) else []


def run_interaction_probe() -> None:
    try:
        deadline = time.monotonic() + 8.0
        before: dict[str, object] | None = None
        while time.monotonic() < deadline:
            try:
                snap = app.debug_snapshot(timeout_ms=1000)
                viewport(snap, LEFT_ID)
                viewport(snap, RIGHT_ID)
                before = snap
                break
            except Exception:
                time.sleep(0.1)
        if before is None:
            raise RuntimeError("timed out waiting for scatter viewport debug details")

        hwnd = None
        for _ in range(30):
            hwnd = find_probe_window()
            if hwnd:
                break
            time.sleep(0.1)
        if not hwnd:
            raise RuntimeError("probe window handle was not found")

        left_center = viewport_center(before, LEFT_ID)
        right_center = viewport_center(before, RIGHT_ID)
        left_view = viewport(before, LEFT_ID)
        right_view = viewport(before, RIGHT_ID)
        left_offset = left_view["offset"]
        left_size = left_view["size"]
        right_offset = right_view["offset"]
        gutter_x = (float(left_offset[0]) + float(left_size[0]) + float(right_offset[0])) * 0.5
        gutter_y = min(left_center[1], right_center[1])

        initial_left = stable_signature(before, LEFT_ID)
        initial_right = stable_signature(before, RIGHT_ID)

        send_mouse_move(hwnd, *left_center)
        send_mouse_move(hwnd, *right_center)
        click(hwnd, *left_center)
        click(hwnd, *right_center)
        click(hwnd, gutter_x, gutter_y)
        drag(hwnd, left_center, (left_center[0] + 36.0, left_center[1] + 20.0))
        drag_outside_window(hwnd, left_center)
        send_mouse_move(hwnd, *right_center)
        drag(hwnd, right_center, (right_center[0] - 24.0, right_center[1] + 12.0))
        wheel(hwnd, *right_center, 120)
        wheel(hwnd, *left_center, -120)
        time.sleep(0.4)

        after = app.debug_snapshot(timeout_ms=2000)
        final_left = stable_signature(after, LEFT_ID)
        final_right = stable_signature(after, RIGHT_ID)
        failures: list[str] = []
        if final_left != initial_left:
            failures.append("left stable scatter signature changed")
        if final_right != initial_right:
            failures.append("right stable scatter signature changed")
        runtime = after.get("runtime", {})
        if isinstance(runtime, dict) and runtime.get("active_scatter_id") is not None:
            failures.append(f"active_scatter_id remained {runtime.get('active_scatter_id')!r}")

        payload = {
            "left": final_left,
            "right": final_right,
            "active_scatter_id": runtime.get("active_scatter_id") if isinstance(runtime, dict) else None,
            "commands": command_tail(after),
        }
