   72: fn points_are_opaque(points: &[PointInstance]) -> bool {
   73:     points.iter().all(|point| point.alpha >= 0.999)
   74: }

   76: fn point_instance_bytes_are_opaque(bytes: &[u8]) -> bool {
   77:     const ALPHA_OFFSET: usize = 28;
   78:     const STRIDE: usize = std::mem::size_of::<PointInstance>();
   79:     bytes.chunks_exact(STRIDE).all(|chunk| {
   80:         let alpha = f32::from_le_bytes([
   81:             chunk[ALPHA_OFFSET],
   82:             chunk[ALPHA_OFFSET + 1],
   83:             chunk[ALPHA_OFFSET + 2],
   84:             chunk[ALPHA_OFFSET + 3],
   85:         ]);
   86:         alpha >= 0.999
   87:     })
   88: }

   90: fn gpu_colormap_uniform(name: &str) -> ([[f32; 4]; MAX_GPU_COLORMAP_POINTS], u32) {
   91:     let points = colormap::resolve(name);
   92:     let mut out = [[0.0; 4]; MAX_GPU_COLORMAP_POINTS];
   93:     let len = points.len().min(MAX_GPU_COLORMAP_POINTS).max(2);
   94:     for i in 0..MAX_GPU_COLORMAP_POINTS {
   95:         let source = points[i.min(points.len().saturating_sub(1))];
   96:         out[i] = [source[0], source[1], source[2], 1.0];
   97:     }
   98:     (out, len as u32)
   99: }

  190: /// Format a scalar bar tick value for display.
  191: /// `value` is always a raw-domain value (matching DragonSci public API).
  192: /// For log-scale bars the value is displayed in scientific notation; for linear bars as decimal.
  193: pub fn format_scalar_bar_tick(value: f32, log_scale: bool) -> String {
  194:     if log_scale {
  195:         format!("{:.2e}", value)
  196:     } else {
  197:         format!("{:.3}", value)
  198:     }
  199: }
  200: 
  201: /// Return up to `count` evenly-spaced tick values between `vmin` and `vmax` (inclusive).
  202: /// For log-scale bars the spacing is even in ln-space but the returned values are raw domain.
  203: /// Returns an empty vec when count < 2; returns a single-element vec when vmin == vmax.
  204: pub fn scalar_bar_tick_values(vmin: f32, vmax: f32, log_scale: bool, count: usize) -> Vec<f32> {
  205:     if count < 2 {
  206:         return Vec::new();
  207:     }
  208:     if vmin == vmax {
  209:         return vec![vmin];
  210:     }
  211:     if log_scale && vmin > 0.0 && vmax > 0.0 {
  212:         let ln_min = vmin.ln();
  213:         let ln_max = vmax.ln();
  214:         (0..count)
  215:             .map(|i| {
  216:                 let t = i as f32 / (count - 1) as f32;
  217:                 (ln_min + t * (ln_max - ln_min)).exp()
  218:             })
  219:             .collect()
  220:     } else {
  221:         (0..count)
  222:             .map(|i| {
  223:                 let t = i as f32 / (count - 1) as f32;
  224:                 vmin + t * (vmax - vmin)
  225:             })
  226:             .collect()
  227:     }
  228: }
