    fn run_result(&self) -> RunResult {
        let frame_ms = if self.frames_rendered > 0 {
            self.frame_ms_total / self.frames_rendered as f64
        } else {
            0.0
        };
        RunResult {
            upload_ms: self.upload_ms,
            frame_ms,
            debug_snapshot: self.debug_snapshot_json(),
        }
    }
