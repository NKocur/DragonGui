 375: def start_stream() -> None:
 376:     if state.stream_thread is not None and state.stream_thread.is_alive():
 377:         set_status("Stream already running", "warning")
 378:         return
 379:     if state.app is None:
 380:         return
 381:     state.stream_stop.clear()
 382: 
 383:     def worker() -> None:
 384:         phase = 0.0
 385:         while not state.stream_stop.is_set():
 386:             phase += 0.23
 387:             sample = build_series(120)
 388:             payload_frame = build_scatter_frame(max(3000, POINTS // 4), phase=phase)
 389: 
 390:             def apply() -> None:
 391:                 state.stream_count += 1
 392:                 if state.stream_led is not None:
 393:                     state.stream_led.set_on(True)
 394:                 if state.progress is not None:
 395:                     state.progress.set_value((state.stream_count % 100) / 100)
 396:                 if state.line_plot is not None:
 397:                     base = state.stream_count * 120
 398:                     xs = np.arange(base, base + 120, dtype=np.float32)
 399:                     ys = np.asarray(sample["latency"], dtype=np.float32)
 400:                     state.line_plot.append_points(xs, ys, series="latency", max_points=900)
 401:                 if state.scatter is not None and state.stream_count % 2 == 0:
 402:                     try:
 403:                         payload = dg.Scatter3D.prepare_points(payload_frame, x="x", y="y", z="z", colormap="turbo")
 404:                         state.scatter.enqueue_prepared_points(payload, coalesce=True, fit=False)
 405:                     except Exception as exc:
 406:                         log(f"Prepared scatter update skipped: {exc}")
 407:                 if state.incident_badge is not None:
 408:                     state.incident_badge.set_value(str(4 + state.stream_count % 5))
 409:                 log(f"Stream tick {state.stream_count}")
 410: 
 411:             try:
 412:                 state.app.call_soon_threadsafe(apply)
 413:             except Exception:
 414:                 break
 415:             state.stream_stop.wait(0.9)
 416: 
 417:         def stopped() -> None:
 418:             if state.stream_led is not None:
 419:                 state.stream_led.set_on(False)
 420:             log("Background stream stopped")
 421: 
 422:         try:
 423:             state.app.call_soon_threadsafe(stopped)
 424:         except Exception:
 425:             pass
 426: 
 427:     state.stream_thread = threading.Thread(target=worker, name="dg-pro-demo-stream", daemon=True)
 428:     state.stream_thread.start()
 429:     set_status("Background stream started", "success")
