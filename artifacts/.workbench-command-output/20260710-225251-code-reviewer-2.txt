 9713:             self.point_size = float(point_size)

 9717:             self.clim = clim

 9747:             t0 = time.perf_counter()
 9748:             payload = self._build_payload()
 9749:             pack_ms = (time.perf_counter() - t0) * 1000.0
 9750:             if payload is None:
 9751:                 raise RuntimeError(

 9753:                 )
 9754:             self._cached_payload = payload
 9755:             self._compute_auto_color_meta()
 9756:             handle.enqueue_set_scatter_points_packed(
