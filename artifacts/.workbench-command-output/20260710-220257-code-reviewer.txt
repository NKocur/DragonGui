 3446: fn scroll_content_bounds_for_child(
 3447:     node: &WidgetNode,
 3448:     result: &LayoutResult,
 3449:     right: &mut f32,
 3450:     top: &mut f32,
 3451:     bottom: &mut f32,
 3452: ) {
 3453:     if is_fixed_positioned_node(node) {

 3506: fn apply_scroll_offsets(
 3507:     root: &WidgetNode,
 3508:     result: &mut LayoutResult,
 3509:     sf: f32,
 3510:     theme: &Theme,
 3511:     state: Option<&WidgetState>,
 3512: ) {
 3513:     let Some(state) = state else {

 3519: fn apply_node_scroll_offsets(
 3520:     node: &WidgetNode,
 3521:     result: &mut LayoutResult,
 3522:     sf: f32,
 3523:     theme: &Theme,
 3524:     state: &WidgetState,
 3525:     inside_scrolled_ancestor: bool,
 3526: ) {
 3527:     if is_scroll_container_node(node) {
 3528:         let use_own_viewport = inside_scrolled_ancestor;
 3529:         let max_scroll_x =
 3530:             scroll_container_max_x_with_viewport(node, result, use_own_viewport, sf, theme);
 3531:         let max_scroll_y =
 3532:             scroll_container_max_y_with_viewport(node, result, use_own_viewport, sf, theme);
 3533:         let scroll_x = state.container_scroll_x(&node.id, max_scroll_x);
 3534:         let scroll_y = state.container_scroll_y(&node.id, max_scroll_y);
 3535:         result.scroll_max_x.insert(node.id.clone(), max_scroll_x);
 3536:         result.scroll_max_y.insert(node.id.clone(), max_scroll_y);
 3537:         result.scroll_x.insert(node.id.clone(), scroll_x);
 3538:         result.scroll_y.insert(node.id.clone(), scroll_y);
 3539:         if scroll_x > 0.0 || scroll_y > 0.0 {
 3540:             for child in &node.children {
 3541:                 if is_fixed_positioned_node(child) {
 3542:                     continue;
 3543:                 }
 3544:                 translate_subtree(child, result, -scroll_x, -scroll_y);
 3545:             }
 3546:         }
 3547:         let inside_scrolled_ancestor =
 3548:             inside_scrolled_ancestor || max_scroll_x > 0.0 || max_scroll_y > 0.0;
 3549:         for child in &node.children {
 3550:             apply_node_scroll_offsets(child, result, sf, theme, state, inside_scrolled_ancestor);
 3551:         }
 3552:         return;
 3553:     }
 3554:     for child in &node.children {
 3555:         apply_node_scroll_offsets(child, result, sf, theme, state, inside_scrolled_ancestor);
 3556:     }
 3557: }
